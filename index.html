// KODE GOOGLE APPS SCRIPT PEKON PANGGUNGREJO UTARA (ANTI-GAGAL)
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Auto-create Sheet Transaksi jika belum ada
    var sheetTrx = ss.getSheetByName("Transaksi");
    if (!sheetTrx) {
      sheetTrx = ss.insertSheet("Transaksi");
      sheetTrx.appendRow(["Tanggal Waktu", "ID Transaksi", "ID Warga", "Nama Warga", "RT", "Nominal", "Petugas"]);
    }
    
    // Auto-create Sheet Warga jika belum ada
    var sheetWarga = ss.getSheetByName("Warga");
    if (!sheetWarga) {
      sheetWarga = ss.insertSheet("Warga");
      sheetWarga.appendRow(["ID Warga", "Kode QR", "Nama Warga", "RT", "No Rumah"]);
    }

    // Ambil Data Transaksi
    var dataTrx = sheetTrx.getDataRange().getValues();
    var transactions = [];
    for (var i = 1; i < dataTrx.length; i++) {
      if (dataTrx[i][0] && dataTrx[i][1]) {
        transactions.push({
          timestamp: dataTrx[i][0] ? new Date(dataTrx[i][0]).toISOString() : new Date().toISOString(),
          id_transaksi: String(dataTrx[i][1]),
          id_warga: String(dataTrx[i][2]),
          nama_warga: String(dataTrx[i][3]),
          rt: String(dataTrx[i][4]),
          nominal: Number(dataTrx[i][5]) || 0,
          petugas: String(dataTrx[i][6]),
          synced: true
        });
      }
    }

    // Ambil Data Warga
    var dataWarga = sheetWarga.getDataRange().getValues();
    var residents = [];
    for (var j = 1; j < dataWarga.length; j++) {
      if (dataWarga[j][0]) {
        residents.push({
          id_warga: String(dataWarga[j][0]),
          kode_qr: String(dataWarga[j][1]),
          nama: String(dataWarga[j][2]),
          rt: String(dataWarga[j][3]),
          no_rumah: String(dataWarga[j][4])
        });
      }
    }

    var result = JSON.stringify({ status: "success", transactions: transactions, residents: residents });
    return ContentService.createTextOutput(result).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    var errorResult = JSON.stringify({ status: "error", message: err.toString() });
    return ContentService.createTextOutput(errorResult).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = {};
    
    if (e && e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      data = e.parameter;
    }

    if (data.action === "add_warga") {
      var sheetWarga = ss.getSheetByName("Warga") || ss.insertSheet("Warga");
      if (sheetWarga.getLastRow() === 0) {
        sheetWarga.appendRow(["ID Warga", "Kode QR", "Nama Warga", "RT", "No Rumah"]);
      }
      sheetWarga.appendRow([
        data.id_warga || "",
        data.kode_qr || "",
        data.nama || "",
        data.rt || "",
        data.no_rumah || ""
      ]);
    } else {
      var sheetTrx = ss.getSheetByName("Transaksi") || ss.insertSheet("Transaksi");
      if (sheetTrx.getLastRow() === 0) {
        sheetTrx.appendRow(["Tanggal Waktu", "ID Transaksi", "ID Warga", "Nama Warga", "RT", "Nominal", "Petugas"]);
      }
      sheetTrx.appendRow([
        data.timestamp ? new Date(data.timestamp) : new Date(),
        data.id_transaksi || ("TRX-" + Date.now()),
        data.id_warga || "",
        data.nama_warga || "",
        data.rt || "",
        Number(data.nominal) || 0,
        data.petugas || ""
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
